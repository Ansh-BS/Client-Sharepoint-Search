// The last few search queries, kept so an empty search box has something
// useful to offer. Queries only — never client names. Cleared on logout,
// because this runs on shared office machines.
const KEY = "sharepoint-search:recents";
const MAX = 3;

function read(): string[] {
  try {
    const raw = localStorage.getItem(KEY);
    const parsed = raw ? JSON.parse(raw) : [];
    if (!Array.isArray(parsed)) return [];
    return parsed.filter((v): v is string => typeof v === "string");
  } catch {
    // Storage disabled, or someone hand-edited the value. Recents are a
    // convenience; never let them break the page.
    return [];
  }
}

function write(list: string[]): void {
  try {
    localStorage.setItem(KEY, JSON.stringify(list));
  } catch {
    /* storage unavailable — carry on without persisting */
  }
}

export function getRecents(): string[] {
  return read();
}

export function addRecent(query: string): string[] {
  const q = query.trim();
  if (!q) return read();
  const rest = read().filter((v) => v.toLowerCase() !== q.toLowerCase());
  const next = [q, ...rest].slice(0, MAX);
  write(next);
  return next;
}

export function clearRecents(): void {
  try {
    localStorage.removeItem(KEY);
  } catch {
    /* storage unavailable — nothing to clear */
  }
}
```

import { useCallback, useEffect, useMemo, useState } from "react";
import Fuse from "fuse.js";
import { SearchField, Button, Card, Link, Spinner } from "@heroui/react";
import { apiGet } from "../lib/api";
import AppHeader from "../components/AppHeader";

export type Client = { id: string; name: string; link: string | null };

const MIN_QUERY = 2;
const MAX_RESULTS = 8;

export default function Search() {
  const [clients, setClients] = useState<Client[] | null>(null);
  const [loadError, setLoadError] = useState(false);
  const [query, setQuery] = useState("");

  // A failed fetch is its own state. Collapsing it into an empty array tells
  // staff there are no clients when the truth is the server never answered.
  const load = useCallback(() => {
    setLoadError(false);
    setClients(null);
    apiGet<Client[]>("/api/clients")
      .then(setClients)
      .catch(() => setLoadError(true));
  }, []);

  useEffect(load, [load]);

  const fuse = useMemo(
    () => new Fuse(clients ?? [], { keys: ["name"], threshold: 0.4 }),
    [clients],
  );

  const trimmed = query.trim();
  const matches = useMemo(() => {
    if (!clients || trimmed.length < MIN_QUERY) return [];
    return fuse.search(trimmed).map((r) => r.item);
  }, [clients, trimmed, fuse]);

  return (
    <div className="min-h-screen bg-background text-foreground p-6">
      <div className="mx-auto max-w-2xl flex flex-col gap-6">
        <AppHeader title="Client SharePoint search" nav={{ to: "/admin", label: "Admin" }} />

        <SearchField aria-label="Search clients" value={query} onChange={setQuery} variant="secondary">
          <SearchField.Group>
            <SearchField.SearchIcon />
            <SearchField.Input placeholder="Search by client name…" />
            <SearchField.ClearButton />
          </SearchField.Group>
        </SearchField>

        {loadError ? (
          <Card variant="secondary">
            <Card.Content className="flex flex-col items-start gap-3">

                        <>
                          <span>{c.name}</span>
                          <span className="text-sm text-muted">No link on file</span>

                    </Card.Content>

                </li>
              ))}
              {matches.length === 0 && <li className="text-muted">No matching clients.</li>}
            </ul>
          </>
        )}
      </div>
    </div>
  );
}
```
